name: "SACL™ Security & Compliance Gate"

on:
  pull_request_target:
    branches: [ "main", "master", "develop" ]
    paths:
      - '.sacl-policy.yml'

permissions:
  contents: read
  pull-requests: write

jobs:
  sacl-compliance-audit:
    name: "SMCF™ Control Tier Verification"
    runs-on: ubuntu-latest
    steps:
      - name: Checkout Secure Governance Tools (Base Branch)
        uses: actions/checkout@11bd71901bbe5b1630ceea73d27597364c9af683 # v4.2.2

      - name: Checkout User Policy Data (Head Fork)
        uses: actions/checkout@11bd71901bbe5b1630ceea73d27597364c9af683 # v4.2.2
        with:
          repository: ${{ github.event.pull_request.head.repo.full_name }}
          ref: ${{ github.event.pull_request.head.ref }}
          sparse-checkout: .sacl-policy.yml
          sparse-checkout-cone-mode: false

      - name: Set Up Python 3.11
        uses: actions/setup-python@42375524e23c412d93fb67b49958b491fce71c38 # v5.4.0
        with:
          python-version: "3.11"

      # 3. Executes script, grabs output, and pushes it safely to $GITHUB_OUTPUT
      - name: Parse User Policy & Run SACL Audit Engine
        id: sacl_audit
        run: |
          python3 -c "
          import yaml, subprocess, sys
          
          with open('.sacl-policy.yml', 'r') as f:
              cfg = yaml.safe_load(f)
          
          swarm = cfg['swarm']
          comp = cfg['compliance']
          
          cmd = [
              'python3', 'tools/sacl_audit.py',
              '--agents', str(swarm['agents']),
              '--ops', str(swarm['ops_per_sec']),
              '--alpha', str(swarm['alpha']),
              '--latency', str(swarm['latency_sec']),
              '--k_cb', str(comp['k_cb_limit'])
          ]
          
          result = subprocess.run(cmd, capture_output=True, text=True)
          
          # Write the console logging back to GitHub environment files
          with open(sys.getenv('GITHUB_OUTPUT'), 'a') as go:
              go.write('audit_results<<EOF\n')
              go.write(result.stdout)
              go.write('\nEOF\n')
              
              # Capture stderr if something broke during the execution
              if result.returncode != 0:
                  go.write('audit_errors<<EOF\n')
                  go.write(result.stderr)
                  go.write('\nEOF\n')
          
          # Print locally for standard actions terminal logs
          print(result.stdout)
          if result.returncode != 0:
              sys.exit(1)
          "

      # 4. Comments the exact dynamic console output back onto the PR chat log
      - name: Comment SMCF™ Audit Status to PR
        uses: actions/github-script@60a0d83039c74a4aee543508d2ffcb1c3799cdea # v7.0.1
        # This ensuring the comment publishes regardless of audit status (success or failure)
        if: always() 
        with:
          script: |
            const status = "${{ steps.sacl_audit.outcome }}" === "success" ? "✅ PASSED" : "❌ FAILED";
            const results = `${{ steps.sacl_audit.outputs.audit_results }}`;
            const errors = `${{ steps.sacl_audit.outputs.audit_errors }}`;
            
            let message = `### 🛡️ Synapticide Maturity & Containment Framework (SMCF)™ Audit Report\n`;
            message += `**Status:** ${status}\n\n`;
            message += `#### 📊 Engine Telemetry Output:\n`;
            message += `\`\`\`text\n${results || 'No output recorded.'}\n\`\`\`\n`;
            
            if (status === "❌ FAILED" && errors) {
              message += `#### ⚠️ Debug Execution Errors:\n`;
              message += `\`\`\`text\n${errors}\n\`\`\`\n`;
            }
            
            message += `\n*Automated evaluation executed via localized governance wrappers.*`;
            
            await github.rest.issues.createComment({
              issue_number: context.issue.number,
              owner: context.repo.owner,
              repo: context.repo.repo,
              body: message
            });
