name: "SACL™ Security & Compliance Gate"

on:
  # This specific trigger is the magic key. It runs using YOUR branch's code, 
  # but allows reading the modified policy file from the contributor's fork.
  pull_request_target:
    branches: [ "main", "master", "develop" ]
    paths:
      - '.sacl-policy.yml' # ONLY trigger if they modified their policy data file

# Strict execution limits. They can read your code and write a PR comment, 
# but they CANNOT write to or modify your repository files.
permissions:
  contents: read
  pull-requests: write

jobs:
  sacl-compliance-audit:
    name: "SMCF™ Control Tier Verification"
    runs-on: ubuntu-latest
    steps:
      # 1. We check out the pristine, secure code from YOUR main branch.
      # This ensures tools/sacl_audit.py is ALWAYS your safe version.
      - name: Checkout Secure Governance Tools (Base Branch)
        uses: actions/checkout@11bd71901bbe5b1630ceea73d27597364c9af683 # v4.2.2

      # 2. We check out ONLY the user's modified .sacl-policy.yml file from their fork.
      - name: Checkout User Policy Data (Head Fork)
        uses: actions/checkout@11bd71901bbe5b1630ceea73d27597364c9af683 # v4.2.2
        with:
          repository: ${{ github.event.pull_request.head.repo.full_name }}
          ref: ${{ github.event.pull_request.head.ref }}
          sparse-checkout: .sacl-policy.yml # DO NOT pull any other files they might have changed
          sparse-checkout-cone-mode: false

      - name: Set Up Python 3.11
        uses: actions/setup-python@42375524e23c412d93fb67b49958b491fce71c38 # v5.4.0
        with:
          python-version: "3.11"

      # 3. Your unchangeable Python script safely parses their data file.
      - name: Parse User Policy & Run SACL Audit Engine
        id: sacl_audit
        run: |
          python3 -c "
          import yaml, subprocess, sys
          
          # Read the data file they provided
          with open('.sacl-policy.yml', 'r') as f:
              cfg = yaml.safe_load(f)
          
          # FIXED: Parsing directly from top-level keys matching your file layout
          swarm = cfg['swarm']
          comp = cfg['compliance']
          
          # Pass their data parameters straight to your pristine local script
          cmd = [
              'python3', 'tools/sacl_audit.py',
              '--agents', str(swarm['agents']),
              '--ops', str(swarm['ops_per_sec']),
              '--alpha', str(swarm['alpha']),
              '--latency', str(swarm['latency_sec']),
              '--k_cb', str(comp['k_cb_limit'])
          ]
          
          result = subprocess.run(cmd, capture_output=True, text=True)
          print(result.stdout)
          
          if result.returncode != 0:
              print('Compliance Audit Failed!')
              sys.exit(1)
          "


      # 4. Record the final results directly to the Pull Request chat log for both of you to see.
      - name: Comment SMCF™ Audit Status to PR
        uses: actions/github-script@60a0d83039c74a4aee543508d2ffcb1c3799cdea # v7.0.1
        with:
          script: |
            const message = `### 🛡️ Synapticide Maturity & Containment Framework (SMCF)™ Audit Report
            **Status:** EVALUATED
            
            The parameters provided in your \`.sacl-policy.yml\` file have been processed by the central governance engine. Please check the workflow execution logs for your full compliance telemetry matrices.`;
            
            github.rest.issues.createComment({
              issue_number: context.issue.number,
              owner: context.repo.owner,
              repo: context.repo.repo,
              body: message
            })
